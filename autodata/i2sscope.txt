@PREFIX=i2sscope
@DEPENDS=I2SAUDIO
@DEVID=I2SSCOPE
@TARGETPREFIX=i2saudio
@TARGET=w_i2saudio
@TRIGGER=w_i2saudio_debug[31]
@LOG_CAPTURE_SIZE=14
@INCLUDEFILE=wbscopc.txt
@INT.I2SSCOPE.PIC=altpic
@INT.I2SSCOPE.WIRE=@$(PREFIX)_int
@$DEFHOLDOFF= (1<<20)-1
@MAIN.DEFNS=
@XDC.INSERT=
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~ *u_@$(TARGETPREFIX)/lli2si/o_lrclk* }] -to [get_cells -hier -filter {NAME=~ *u_@$(TARGETPREFIX)/xclk_debug* }] 8.0
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~ *u_@$(TARGETPREFIX)/lli2si/o_bclk* }] -to [get_cells -hier -filter {NAME=~ *u_@$(TARGETPREFIX)/xclk_debug* }] 8.0
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~ *u_@$(TARGETPREFIX)/lli2si/o_dac* }] -to [get_cells -hier -filter {NAME=~ *u_@$(TARGETPREFIX)/xclk_debug* }] 8.0
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~ *xgenclki/GEN_PLL.pll* }] -to [get_cells -hier -filter {NAME=~ *u_i2saudio/xclk_debug* }] 8.0
