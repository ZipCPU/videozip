@PREFIX=vidscope
@DEVID=VIDSCOPE
@DEPENDS=VIDPIPE_ACCESS
@TARGET=vidpipe
@$LOG_CAPTURE_SIZE=9
@MAIN.INSERT=
	@$(CORE) #(
		// {{{
		.LGMEM(@$(LOG_CAPTURE_SIZE)),
		.SYNCHRONOUS(@$(SYNCHRONOUS)),
		.DEFAULT_HOLDOFF(@$(DEFHOLDOFF))
		// }}}
	) @$(PREFIX)i(
		// {{{
		.i_data_clk(@$(DATA_CLOCK)), .i_ce(@$(CAPTURECE)),
		.i_trigger(@$(TRIGGER)), .i_data(@$(DEBUG)),
		.i_wb_clk(@$(SLAVE.BUS.CLOCK.WIRE)),
		@$(SLAVE.ANSIPORTLIST),
		.o_interrupt(@$(PREFIX)_int)
		// }}}
	);

@MAIN.DEFNS=
@TRIGGER=hdmidbg_trigger
@CAPTURECE=hdmidbg_ce
@DEBUG=hdmiclr_debug
@INT.VIDSCOPE.WIRE=@$(PREFIX)_int
@INT.VIDSCOPE.PIC=altpic
@DATA_CLOCK=i_pixclk
@SYNCHRONOUS=0
@INCLUDEFILE=wbscope.txt
