set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~ thedesign/vidscopei/GEN_ASYNC.r_reset_complete*}] -to [get_cells -hier -filter {NAME=~thedesign/vidscopei/GEN_ASYNC.q_reset_complete*}] 5.0
