@PREFIX=mem_sdram_only
@MASTER.TYPE=SCRIPT
@MASTER.BUS=wbwide
@LD.FILE=sdram.ld
@LD.DEFNS=
/* LD.DEFNS from zipmaster.txt */
_kram  = 0; /* No high-speed kernel RAM */
_ram   = ORIGIN(sdram);
_rom   = 0;
_top_of_stack = ORIGIN(sdram) + LENGTH(sdram);

@LD.SCRIPT=
/* LD.SCRIPT from zipmaster.txt */
SECTIONS
{
       .rocode ORIGIN(sdram) : ALIGN(4) {
               _boot_address = .;
               *(.start) *(.boot)
               } > sdram
       _kram_start = . ;
       _kram_end = . ;
       _ram_image_start = . ;
       .kernel : ALIGN_WITH_INPUT {
               *(.kernel)
               *(.text.startup)
               *(.text*)
               *(.rodata*) *(.strings)
               *(.data) *(COMMON)
               }> sdram
       _ram_image_end = . ;
       .bss : ALIGN_WITH_INPUT {
               *(.bss)
               _bss_image_end = . ;
               } > sdram
       _top_of_heap = .;
}
