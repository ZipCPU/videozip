@PREFIX=genclk
@ACCESS=ARBITRARY_CLOCK_GENERATOR_ACCESS
@DEVID=GENCLK
@NADDR=1
@SLAVE.TYPE=SINGLE
@SLAVE.BUS=wb32
@TOP.PORTLIST=
		o_unused_fan_pwm
@TOP.IODECL=
	inout	wire	o_unused_fan_pwm;
@TOP.DEFNS=
	////////////////////////////////////////////////////////////////////////
	//
	// Definitions for the clock generation circuit
	// {{{
	wire		s_@$(PREFIX)_clk, w_@$(PREFIX)_pll_locked,
			w_@$(PREFIX)_ce;
	wire	[7:0]	w_@$(PREFIX)_word;
	// }}}
@TOP.INSERT=
	////////////////////////////////////////////////////////////////////////
	//
	// (Audio) clock generation logic
	// {{{

	xgenclk
	x@$(PREFIX)i(
		// {{{
		s_clksync, s_clk_400mhz,
			w_@$(PREFIX)_ce, w_@$(PREFIX)_word,
			o_unused_fan_pwm,
			s_@$(PREFIX)_clk, w_@$(PREFIX)_pll_locked
		// }}}
	);

	// }}}
@TOP.MAIN=
		// Clock Generator ports
		s_@$(PREFIX)_clk,
		w_@$(PREFIX)_pll_locked,
		w_@$(PREFIX)_word,
		w_@$(PREFIX)_ce
@MAIN.PORTLIST=
		// Clock generator ports
		i_@$(PREFIX)_clk,
		i_@$(PREFIX)_pll_locked,
		o_@$(PREFIX)_word,
		o_@$(PREFIX)_ce
@MAIN.IODECL=
	// Clock generator I/O ports
	// {{{
	input	wire		i_@$(PREFIX)_clk;
	input	wire		i_@$(PREFIX)_pll_locked;
	output	wire	[7:0]	o_@$(PREFIX)_word;
	output	wire		o_@$(PREFIX)_ce;
	// }}}
@MAIN.DEFNS=
	////////////////////////////////////////////////////////////////////////
	//
	// Clock generator signal definitions
	// {{{
	// Verilator lint_off UNUSED
	// Set for 24.576 MHz
	localparam [39:0] DEF_GENCLK_FREQUENCY = (1<<23) * 24_576 / 3_125;
	// Verilator lint_on  UNUSED
	reg	[30:0]	r_@$(PREFIX)_data;
	wire		w_@$(PREFIX)_unused_stb;
	reg		r_@$(PREFIX)_ack;
	// }}}
@MAIN.INSERT=
	////////////////////////////////////////////////////////////////////////
	//
	// Generic clock frequency generator
	// {{{
	initial	r_@$(PREFIX)_data = { 1'b1, DEF_GENCLK_FREQUENCY[29:0] };

	always @(posedge @$(SLAVE.BUS.CLOCK.WIRE))
	if (i_reset)
		r_@$(PREFIX)_data <= { 1'b1, DEF_GENCLK_FREQUENCY[29:0] };
	else if (@$(SLAVE.PREFIX)_stb && @$(SLAVE.PREFIX)_we)
		r_@$(PREFIX)_data <= @$(SLAVE.PREFIX)_data[30:0];

	always @(posedge @$(SLAVE.BUS.CLOCK.WIRE))
		r_@$(PREFIX)_ack <= @$(SLAVE.PREFIX)_stb;
	assign	@$(SLAVE.PREFIX)_ack   = r_@$(PREFIX)_ack;
	assign	@$(SLAVE.PREFIX)_stall = 1'b0;
	assign	@$(SLAVE.PREFIX)_idata = { i_@$(PREFIX)_pll_locked, r_@$(PREFIX)_data };
	assign	o_@$(PREFIX)_ce = r_@$(PREFIX)_data[30];

	genclk
	u_@$(PREFIX)(
		@$(SLAVE.BUS.CLOCK.WIRE),
		{ 2'b00, r_@$(PREFIX)_data[29:0] },
		o_@$(PREFIX)_word,
		w_@$(PREFIX)_unused_stb
	);
@RTL.MAKE.GROUP=GENCLK
@RTL.MAKE.SUBD=
@RTL.MAKE.FILES=genclk.v xgenclk.v
@REGS.N=1
@REGS.0= 0 R_@$(DEVID) @$(DEVID)
