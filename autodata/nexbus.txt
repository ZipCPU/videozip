@PREFIX=netbus
@DEVID=NETBUS
@ACCESS=NETBUS_ACCESS
@DEPENDS=MEGANET_ACCESS
@MASTER.BUS=wbu
@MASTER.TYPE=HOST
@MASTER.PREFIX=@$(PREFIX)
@CLOCK.NAME=clk
@CLOCK.RESET=i_reset
@$BUS_ADDRESS_WIDTH=@$(MASTER.BUS.AWID)
@NET=net
@NETADDR=net_ip_addr
@NETVALID=net_dbg_valid
@NETREADY=net_dbg_ready
@NETDATA=net_dbg_data
@NETBYTES=net_dbg_bytes
@NETLAST=net_dbg_last
@MAIN.PARAM=
	////////////////////////////////////////////////////////////////////////
	//
	// NETBUS parameters
	// {{{
	localparam	@$(DEVID)WATCHDOG_RAW = 5 + 9;
	localparam	@$(DEVID)WATCHDOG = (@$(DEVID)WATCHDOG_RAW > 19)
				? @$(DEVID)WATCHDOG_RAW : 19;
	// }}}
@MAIN.DEFNS=
	////////////////////////////////////////////////////////////////////////
	//
	// NETBUS signals
	// {{{
	// Verilator lint_off UNUSED
	wire	[29:0]	@$(MASTER.PREFIX)_tmp_addr;
	wire	[9:0]	@$(PREFIX)_gpout;
	// Verilator lint_on  UNUSED

	wire		@$(PREFIX)_valid, @$(PREFIX)_ready, @$(PREFIX)_last;
	wire	[31:0]	@$(PREFIX)_pkdata;
	// wire	[1:0]	@$(PREFIX)_bytes;
	wire		w_@$(PREFIX)_int; // 1'b0;
	// Verilator lint_off UNUSED
	wire	[31:0]	@$(PREFIX)_debug;
	// Verilator lint_on  UNUSED
	// }}}
@MAIN.INSERT=
	////////////////////////////////////////////////////////////////////////
	//
	// NETBUS: UDP driven bus master
	// {{{
`ifdef	BUSPIC_ACCESS
	assign	w_@$(PREFIX)_int = w_bus_int;
`else
	assign	w_@$(PREFIX)_int = 1'b0;
`endif

	netdebug #(
		// {{{
		.LGWATCHDOG(@$(DEVID)WATCHDOG),
		.DBG_UDPPORT(UDP_DBGPORT)
		// }}}
	) u_@$(PREFIX) (
		// {{{
		.i_clk(@$(CLOCK.WIRE)), .i_reset(@$(CLOCK.RESET)),
		// Config
		// {{{
		.i_gpio({ 9'h0, w_@$(PREFIX)_int }), .o_gpio(@$(PREFIX)_gpout),
		.i_my_ipaddr(@$(NETADDR)),
		.i_interrupt(w_@$(PREFIX)_int),
		// }}}
		// Wishbone
		// {{{
		.o_wb_cyc(@$(MASTER.PREFIX)_cyc), .o_wb_stb(@$(MASTER.PREFIX)_stb),
			.o_wb_we(@$(MASTER.PREFIX)_we),
			.o_wb_addr(@$(MASTER.PREFIX)_tmp_addr),
			.o_wb_data(@$(MASTER.PREFIX)_data),
			.o_wb_sel(@$(MASTER.PREFIX)_sel),
		.i_wb_stall(@$(MASTER.PREFIX)_stall),
			.i_wb_ack(@$(MASTER.PREFIX)_ack),
		.i_wb_data(@$(MASTER.PREFIX)_idata),
		.i_wb_err(@$(MASTER.PREFIX)_err),
		// }}}
		// Incoming network data
		// {{{
		.S_AXI_VALID(@$(NETVALID)), .S_AXI_READY(@$(NETREADY)),
			.S_AXI_DATA(@$(NETDATA)),
			.S_AXI_BYTES(@$(NETBYTES)),
			.S_AXI_LAST(@$(NETLAST)),
		// }}}
		// Return network data
		// {{{
		.M_AXI_VALID(@$(PREFIX)_valid),
			.M_AXI_READY(@$(PREFIX)_ready),
			.M_AXI_DATA(@$(PREFIX)_pkdata),
			// BYTES=0, all words are full
			.M_AXI_LAST(@$(PREFIX)_last),
			// Not abortable here
		// }}}
		.o_debug(@$(PREFIX)_debug)
		// }}}
	);

	assign	@$(MASTER.PREFIX)_addr = @$(MASTER.PREFIX)_tmp_addr[(@$BUS_ADDRESS_WIDTH-1):0];
	// }}}
@MAIN.ALT=
	assign	@$(PREFIX)_valid = 1'b0;
	assign	@$(PREFIX)_pkdata  = 32'h0;
	// assign	@$(PREFIX)_bytes = 2'b0;
	assign	@$(PREFIX)_last  = 1'b1;
	assign	@$(PREFIX)_abort  = 1'b0;
@XDC.INSERT=
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~ thedesign/u_@$(NET)/dbgtx_afifo/mem*}] -to [get_cells -hier -filter {NAME=~ thedesign/u_@$(NET)/dbgtx_afifo/GEN_REGISTERED_READ.o_rd_data*}] 8.0
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~ thedesign/u_@$(NET)/dbg_afifo/mem*}] -to [get_cells -hier -filter {NAME=~ thedesign/u_@$(NET)/dbg_afifo/GEN_REGISTERED_READ.o_rd_data*}] 8.0
@RTL.MAKE.GROUP= NETBUS
@RTL.MAKE.SUBD=proto
@RTL.MAKE.FILES= netdebug.v netdbgrx.v netdbgtx.v
